# Component Patterns

Advanced component patterns used in shadcn-storybook-registry.

## Compound Components

The **Compound Components** pattern is one in which several components implicitly share state.

### Concept

The parent component provides state to its child components through Context:

```typescript
// ✅ Compound Components pattern
<Dialog>
  <DialogTrigger>Open</DialogTrigger>
  <DialogContent>
    <DialogHeader>
      <DialogTitle>Title</DialogTitle>
      <DialogDescription>Description</DialogDescription>
    </DialogHeader>
  </DialogContent>
</Dialog>
```

### Implementation Example: Dialog

```typescript
import * as React from "react";

// Share state via Context
const DialogContext = React.createContext<{
  open: boolean;
  setOpen: (open: boolean) => void;
}>({
  open: false,
  setOpen: () => {},
});

// Parent component
function Dialog({ children }: { children: React.ReactNode }) {
  const [open, setOpen] = React.useState(false);

  return (
    <DialogContext.Provider value={{ open, setOpen }}>
      {children}
    </DialogContext.Provider>
  );
}

// Child components consume the Context
function DialogTrigger({ children }: { children: React.ReactNode }) {
  const { setOpen } = React.useContext(DialogContext);
  return <button onClick={() => setOpen(true)}>{children}</button>;
}

function DialogContent({ children }: { children: React.ReactNode }) {
  const { open, setOpen } = React.useContext(DialogContext);
  if (!open) return null;
  return (
    <div>
      {children}
      <button onClick={() => setOpen(false)}>Close</button>
    </div>
  );
}
```

### Benefits

- ✅ **Implicit state sharing**: Eliminates props drilling
- ✅ **Flexible composition**: Component order and structure can be changed freely
- ✅ **Readability**: The component hierarchy is clear

### shadcn/ui Use Cases

The following components use the Compound Components pattern:

- **Dialog**: `<Dialog>`, `<DialogTrigger>`, `<DialogContent>`, `<DialogHeader>`
- **Popover**: `<Popover>`, `<PopoverTrigger>`, `<PopoverContent>`
- **Accordion**: `<Accordion>`, `<AccordionItem>`, `<AccordionTrigger>`, `<AccordionContent>`
- **Tabs**: `<Tabs>`, `<TabsList>`, `<TabsTrigger>`, `<TabsContent>`
- **Select**: `<Select>`, `<SelectTrigger>`, `<SelectContent>`, `<SelectItem>`

## Render Props

The **Render Props** pattern delegates rendering by passing a function as a prop.

### Concept

The component receives its rendering logic from outside:

```typescript
// ✅ Render Props pattern
<DataProvider
  render={(data) => (
    <div>
      {data.map((item) => (
        <Card key={item.id}>{item.name}</Card>
      ))}
    </div>
  )}
/>
```

### Implementation Example: DataProvider

```typescript
interface DataProviderProps<T> {
  data: T[];
  render: (data: T[]) => React.ReactNode;
  loading?: (isLoading: boolean) => React.ReactNode;
}

function DataProvider<T>({ data, render, loading }: DataProviderProps<T>) {
  const [isLoading, setIsLoading] = React.useState(true);
  const [items, setItems] = React.useState<T[]>([]);

  React.useEffect(() => {
    // Simulate data loading
    setTimeout(() => {
      setItems(data);
      setIsLoading(false);
    }, 1000);
  }, [data]);

  if (isLoading && loading) {
    return <>{loading(isLoading)}</>;
  }

  return <>{render(items)}</>;
}

// Usage example
<DataProvider
  data={users}
  loading={(isLoading) => <Skeleton />}
  render={(users) => (
    <div>
      {users.map((user) => (
        <UserCard key={user.id} user={user} />
      ))}
    </div>
  )}
/>;
```

### Children as Function

A variant pattern that receives `children` as a function:

```typescript
interface ToggleProps {
  children: (isOn: boolean, toggle: () => void) => React.ReactNode;
}

function Toggle({ children }: ToggleProps) {
  const [isOn, setIsOn] = React.useState(false);
  const toggle = () => setIsOn(!isOn);

  return <>{children(isOn, toggle)}</>;
}

// Usage example
<Toggle>
  {(isOn, toggle) => (
    <Button onClick={toggle}>{isOn ? "ON" : "OFF"}</Button>
  )}
</Toggle>;
```

### Benefits

- ✅ **Reusability**: Separates logic from rendering
- ✅ **Flexibility**: Implements a variety of UIs with the same logic
- ✅ **Type Safety**: Ensures type safety with TypeScript generics

## Custom Hooks

**Custom Hooks** is a pattern for encapsulating reusable logic.

### Concept

Component logic is extracted into a function and reused:

```typescript
// ✅ Custom Hook pattern
function useDialog() {
  const [open, setOpen] = React.useState(false);

  const openDialog = () => setOpen(true);
  const closeDialog = () => setOpen(false);
  const toggleDialog = () => setOpen(!open);

  return { open, openDialog, closeDialog, toggleDialog };
}

// Usage example
function MyComponent() {
  const dialog = useDialog();

  return (
    <>
      <Button onClick={dialog.openDialog}>Open Dialog</Button>
      <Dialog open={dialog.open} onOpenChange={dialog.setOpen}>
        <DialogContent>Hello</DialogContent>
      </Dialog>
    </>
  );
}
```

### Implementation Example: useToggle

```typescript
function useToggle(initialValue = false) {
  const [value, setValue] = React.useState(initialValue);

  const toggle = React.useCallback(() => {
    setValue((v) => !v);
  }, []);

  const setTrue = React.useCallback(() => {
    setValue(true);
  }, []);

  const setFalse = React.useCallback(() => {
    setValue(false);
  }, []);

  return { value, toggle, setTrue, setFalse };
}
```

### Implementation Example: useLocalStorage

```typescript
function useLocalStorage<T>(key: string, initialValue: T) {
  // Load the initial value
  const [storedValue, setStoredValue] = React.useState<T>(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(error);
      return initialValue;
    }
  });

  // Save the value
  const setValue = (value: T | ((val: T) => T)) => {
    try {
      const valueToStore =
        value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      window.localStorage.setItem(key, JSON.stringify(valueToStore));
    } catch (error) {
      console.error(error);
    }
  };

  return [storedValue, setValue] as const;
}

// Usage example
function ThemeToggle() {
  const [theme, setTheme] = useLocalStorage("theme", "light");

  return (
    <Button onClick={() => setTheme(theme === "light" ? "dark" : "light")}>
      {theme === "light" ? "🌙" : "☀️"}
    </Button>
  );
}
```

## Controlled vs Uncontrolled

The **Controlled** and **Uncontrolled** component patterns.

### Controlled Components

State is controlled by the parent component:

```typescript
function ControlledInput() {
  const [value, setValue] = React.useState("");

  return (
    <Input
      value={value}
      onChange={(e) => setValue(e.target.value)}
    />
  );
}
```

### Uncontrolled Components

State is controlled by the DOM (using `ref`):

```typescript
function UncontrolledInput() {
  const inputRef = React.useRef<HTMLInputElement>(null);

  const handleSubmit = () => {
    console.log(inputRef.current?.value);
  };

  return (
    <>
      <Input ref={inputRef} defaultValue="Hello" />
      <Button onClick={handleSubmit}>Submit</Button>
    </>
  );
}
```

### Hybrid Pattern

shadcn/ui uses a Hybrid pattern that **supports both Controlled and Uncontrolled** usage:

```typescript
interface DialogProps {
  open?: boolean; // Controlled
  defaultOpen?: boolean; // Uncontrolled
  onOpenChange?: (open: boolean) => void;
}

function Dialog({ open, defaultOpen, onOpenChange }: DialogProps) {
  const [uncontrolledOpen, setUncontrolledOpen] = React.useState(
    defaultOpen ?? false
  );

  // Use open when Controlled; use internal state when Uncontrolled
  const isOpen = open !== undefined ? open : uncontrolledOpen;

  const handleOpenChange = (newOpen: boolean) => {
    if (open === undefined) {
      setUncontrolledOpen(newOpen);
    }
    onOpenChange?.(newOpen);
  };

  return (
    // Dialog implementation
    <div>{isOpen ? "Open" : "Closed"}</div>
  );
}

// Controlled usage
<Dialog open={open} onOpenChange={setOpen} />

// Uncontrolled usage
<Dialog defaultOpen={false} />
```

### Selection Criteria

| Pattern | When to Use | Example |
|---------|----------|------|
| **Controlled** | When the parent must manage state | Form validation, synchronisation |
| **Uncontrolled** | Simple inputs, performance optimisation | Simple forms, file inputs |
| **Hybrid** | When you want to support both | shadcn/ui components |

## Higher-Order Components (HOC)

An **HOC** is a function that takes a component as an argument and returns a new component.

### Concept

```typescript
// HOC function
function withLoading<P extends object>(
  Component: React.ComponentType<P>
) {
  return function WithLoadingComponent({
    isLoading,
    ...props
  }: P & { isLoading: boolean }) {
    if (isLoading) {
      return <Skeleton />;
    }
    return <Component {...(props as P)} />;
  };
}

// Usage example
const UserCardWithLoading = withLoading(UserCard);

<UserCardWithLoading isLoading={true} user={user} />;
```

### Implementation Example: withAuth

```typescript
function withAuth<P extends object>(
  Component: React.ComponentType<P>
) {
  return function WithAuthComponent(props: P) {
    const { user, isAuthenticated } = useAuth();

    if (!isAuthenticated) {
      return <LoginPage />;
    }

    return <Component {...props} user={user} />;
  };
}

// Usage example
const ProtectedDashboard = withAuth(Dashboard);
```

### Caveats

- ⚠️ **In modern React, replacing HOCs with Custom Hooks is recommended**
- ⚠️ Ref forwarding is required
- ⚠️ Harder to debug

```typescript
// ❌ Instead of an HOC
const UserCardWithLoading = withLoading(UserCard);

// ✅ Use a Custom Hook
function UserCardWithLoading(props) {
  const { isLoading } = useUserData();
  if (isLoading) return <Skeleton />;
  return <UserCard {...props} />;
}
```

## Composition Pattern

The **Composition** pattern builds complex UIs by combining small components.

### Concept

```typescript
// ✅ Composition pattern
<Card>
  <CardHeader>
    <CardTitle>Title</CardTitle>
    <CardDescription>Description</CardDescription>
  </CardHeader>
  <CardContent>
    <p>Content</p>
  </CardContent>
  <CardFooter>
    <Button>Action</Button>
  </CardFooter>
</Card>
```

### Benefits

- ✅ **Reusability**: Combine small components in many ways
- ✅ **Flexibility**: Compose only the parts you need
- ✅ **Readability**: The component structure is clear

### Implementation Example: Card

```typescript
function Card({ children, ...props }: React.HTMLAttributes<HTMLDivElement>) {
  return (
    <div className="rounded-lg border bg-card text-card-foreground shadow-sm" {...props}>
      {children}
    </div>
  );
}

function CardHeader({ children, ...props }: React.HTMLAttributes<HTMLDivElement>) {
  return (
    <div className="flex flex-col space-y-1.5 p-6" {...props}>
      {children}
    </div>
  );
}

function CardTitle({ children, ...props }: React.HTMLAttributes<HTMLHeadingElement>) {
  return (
    <h3 className="text-2xl font-semibold leading-none tracking-tight" {...props}>
      {children}
    </h3>
  );
}

// Usage example: compose only the parts you need
<Card>
  <CardHeader>
    <CardTitle>Only Title</CardTitle>
  </CardHeader>
</Card>

<Card>
  <CardContent>
    <p>Only Content</p>
  </CardContent>
  <CardFooter>
    <Button>Action</Button>
  </CardFooter>
</Card>
```

## Polymorphic Components

**Polymorphic Components** is a pattern that dynamically changes the rendered HTML tag via the `as` prop.

### Concept

```typescript
// ✅ Polymorphic Component
<Button as="a" href="/login">
  Login
</Button>

<Button as="div" onClick={handleClick}>
  Div Button
</Button>
```

### Implementation Example

```typescript
type PolymorphicProps<E extends React.ElementType> = {
  as?: E;
  children: React.ReactNode;
} & React.ComponentPropsWithoutRef<E>;

function Button<E extends React.ElementType = "button">({
  as,
  children,
  ...props
}: PolymorphicProps<E>) {
  const Component = as || "button";
  return (
    <Component className="btn" {...props}>
      {children}
    </Component>
  );
}

// Usage example
<Button>Default Button</Button>
<Button as="a" href="/home">Link Button</Button>
<Button as="div" role="button">Div Button</Button>
```

### Benefits

- ✅ **Flexibility**: Use a variety of HTML tags with the same styling
- ✅ **Type Safety**: Prop types are inferred automatically by TypeScript
- ✅ **Accessibility**: Allows the correct semantic HTML to be chosen

## Best Practices

### 1. Prefer Compound Components

Use Compound Components when complex state sharing is required:

```typescript
// ✅ Good: Compound Components
<Select>
  <SelectTrigger>Select</SelectTrigger>
  <SelectContent>
    <SelectItem value="1">Option 1</SelectItem>
  </SelectContent>
</Select>

// ❌ Bad: Passing all props to the parent
<Select
  trigger={<button>Select</button>}
  content={[{ value: "1", label: "Option 1" }]}
/>
```

### 2. Separate Logic with Custom Hooks

Extract reusable logic into a Custom Hook:

```typescript
// ✅ Good: Custom Hook
function useFormValidation(schema) {
  const [errors, setErrors] = React.useState({});
  const validate = (data) => {
    // validation logic
  };
  return { errors, validate };
}

// ❌ Bad: Logic duplicated inside components
function Form1() {
  const [errors, setErrors] = React.useState({});
  // duplicated validation logic
}
```

### 3. Use TypeScript Generics

Use generics for type safety:

```typescript
// ✅ Good: Ensure type safety with generics
function useData<T>(fetcher: () => Promise<T>) {
  const [data, setData] = React.useState<T | null>(null);
  // ...
  return data;
}

// ❌ Bad: Using any
function useData(fetcher: () => Promise<any>) {
  // ...
}
```
